from importlib.metadata import version
import torch
import torchaudio

packages = (
    "numpy",
    "torch",
    "torchaudio",
    "pyannote.audio",
    "pyannote.core",
    "pyannote.database",
    "pyannote.metrics",
    "pyannote.pipeline",
)

for package in packages:
    print(package, version(package))

print("CUDA:", torch.version.cuda)
print("torchaudio.info:", hasattr(torchaudio, "info"))

assert version("numpy") == "1.26.4"
assert version("pyannote.audio") == "3.4.0"
assert version("torch").startswith("2.8.0")
assert version("torchaudio").startswith("2.8.0")
assert torch.version.cuda is None
assert hasattr(torchaudio, "info")
